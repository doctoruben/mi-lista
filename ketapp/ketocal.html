async function analizarIA() { const key = localStorage.getItem('GEMINI_KEY'); const text = document.getElementById('foodInput').value.trim(); const status = document.getElementById('status'); if(!key) { alert("Configura la Key en ⚙️"); return; } if(!text) return; status.innerText = "Analizando con motor 2.5..."; document.getElementById('resultCard').style.display = 'none'; try { const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${key}`; const response = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: `Analiza: "${text}". Devuelve exclusivamente un objeto JSON: {"n": "nombre plato", "c": carbohidratos_numero, "k": calorias_numero}` }] }] }) }); const data = await response.json(); // VALIDACIÓN DE SEGURIDAD PARA EVITAR EL ERROR 'UNDEFINED' if (!data.candidates || data.candidates.length === 0 || !data.candidates[0].content) { if (data.error) { throw new Error(data.error.message); } else { throw new Error("La IA no ha podido generar una respuesta. Inténtalo de nuevo."); } } let txt = data.candidates[0].content.parts[0].text; // LIMPIEZA ROBUSTA DEL JSON const inicio = txt.indexOf('{'); const fin = txt.lastIndexOf('}'); if (inicio === -1 || fin === -1) throw new Error("Formato de respuesta inválido"); const jsonLimpio = txt.substring(inicio, fin + 1); tempFood = JSON.parse(jsonLimpio); document.getElementById('resName').innerText = tempFood.n; document.getElementById('resNutri').innerText = `${tempFood.c}g Carbs | ${tempFood.k} Kcal`; document.getElementById('resultCard').style.display = 'block'; status.innerText = ""; } catch (err) { console.error("Detalle del error:", err); status.innerText = "Error: " + err.message; } }
